cmake_minimum_required(VERSION 3.26)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    # Respect explicit architectures before CUDA compiler detection.
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES AND NOT DEFINED ENV{CUDAARCHS})
        set(CMAKE_CUDA_ARCHITECTURES native)
    endif()
    project(tyloinfer_d1_native LANGUAGES CXX CUDA)
    set(TYLOINFER_BUILD_RUNTIME_COMMUNICATION OFF)
    set(TYLOINFER_ENABLE_MFQ_COMPAT OFF)
    add_compile_definitions(TYLOINFER_ENABLE_MFQ_COMPAT=0)
endif()
get_filename_component(_d1_repository "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
if(NOT DEFINED TYLOINFER_REPOSITORY_ROOT)
    set(TYLOINFER_REPOSITORY_ROOT "${_d1_repository}")
endif()
set(_runtime "${_d1_repository}/csrc")
if(NOT TARGET mfq-json)
    add_subdirectory("${_runtime}/components/json" "components/json")
endif()
if(NOT TARGET mfq-ggml)
    add_subdirectory("${_runtime}/components/ggml" "components/ggml")
endif()
if(NOT TARGET mfq-model-source)
    add_subdirectory("${_runtime}/core" "core")
endif()
find_package(CUDAToolkit 11.2 REQUIRED)
add_executable(tyloinfer-d1-native main.cpp input.cpp engine.cu "${_runtime}/backends/cuda/kernels/gguf_gemm.cu")
target_include_directories(tyloinfer-d1-native PRIVATE "${_runtime}" "${_runtime}/components/ggml/src" "${_runtime}/core/include")
target_link_libraries(tyloinfer-d1-native PRIVATE mfq-model-source mfq-ggml mfq-json CUDA::cudart CUDA::cublas)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    set(_d1_output "${CMAKE_BINARY_DIR}/bin")
else()
    set(_d1_output "${PROJECT_BINARY_DIR}")
endif()
set_target_properties(tyloinfer-d1-native PROPERTIES
    CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON
    RUNTIME_OUTPUT_DIRECTORY "${_d1_output}")
if(MSVC)
    target_compile_options(tyloinfer-d1-native PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/utf-8>" "$<$<COMPILE_LANGUAGE:CXX>:/EHsc>")
endif()
